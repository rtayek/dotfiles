mkdir -p ~/dotfiles/direnv

cat > ~/dotfiles/direnv/envrc <<'EOF'
# Shared direnv library (sourced by workspace/project .envrc files)
# Keep this fast, deterministic, and non-interactive.

pathAdd() {
  # Usage: pathAdd <dir>
  # Prepends <dir> to PATH if it exists and isn't already present.
  local d="$1"
  [ -n "$d" ] || return 0
  [ -d "$d" ] || return 0
  case ":$PATH:" in
    *":$d:"*) ;;
    *) export PATH="$d:$PATH" ;;
  esac
}

isGitRepo() { [ -d .git ] || git rev-parse --is-inside-work-tree >/dev/null 2>&1; }
isGradleProject() { [ -f build.gradle ] || [ -f build.gradle.kts ] || [ -f settings.gradle ] || [ -f settings.gradle.kts ]; }
isMavenProject() { [ -f pom.xml ]; }

loadSecretsIfPresent() {
  # Usage: loadSecretsIfPresent [file]
  # Default: .env.secrets (untracked/ignored)
  local f="${1:-.env.secrets}"
  [ -f "$f" ] || return 0
  # shellcheck disable=SC1090
  source "$f"
}

useProjectHistory() {
  # Per-directory history file + force bash to switch history backing file now.
  export HISTFILE="$PWD/.bash_history"
  history -a
  history -c
  history -r
}

useJava() {
  # Usage: useJava <javaHomePath>
  # Windows Git Bash example: useJava "/c/Java/jdk-25"
  local jh="$1"
  [ -n "$jh" ] || return 0
  [ -d "$jh" ] || { echo "direnv: JAVA_HOME not found: $jh" >&2; return 1; }
  export JAVA_HOME="$jh"
  pathAdd "$JAVA_HOME/bin"
}

useJavaFromFile() {
  # Usage: useJavaFromFile [file]
  # File contains a single path, e.g. /c/Java/jdk-25
  local f="${1:-.java-home}"
  [ -f "$f" ] || return 0
  local jh
  jh="$(tr -d '\r\n' < "$f")"
  useJava "$jh"
}

useGradleCacheLocal() {
  # Keep Gradle noise inside the project (good for determinism / clean global state)
  export GRADLE_USER_HOME="$PWD/.gradle"
}

useGradleCacheShared() {
  # Usage: useGradleCacheShared <dir>
  # Example: useGradleCacheShared "$HOME/.gradle-cache-shared"
  local d="$1"
  [ -n "$d" ] || return 0
  mkdir -p "$d" 2>/dev/null || true
  export GRADLE_USER_HOME="$d"
}

useNodeBin() {
  # If a project has node_modules/.bin, prefer it.
  pathAdd "$PWD/node_modules/.bin"
}

requireFile() {
  # Usage: requireFile <path> <message>
  local f="$1"
  local msg="${2:-Missing required file: $f}"
  [ -f "$f" ] || { echo "direnv: $msg" >&2; return 1; }
}

requireDir() {
  # Usage: requireDir <path> <message>
  local d="$1"
  local msg="${2:-Missing required directory: $d}"
  [ -d "$d" ] || { echo "direnv: $msg" >&2; return 1; }
}
EOF
